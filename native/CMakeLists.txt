cmake_minimum_required(VERSION 3.24)
project(LevelGeometry LANGUAGES CXX)
add_library(level_geometry SHARED geometry.cpp)
target_compile_features(level_geometry PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(level_geometry PRIVATE /utf-8 /W4 /O2)
  set_property(TARGET level_geometry PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set_target_properties(level_geometry PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../bin" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/../bin")
